### Compiled fct_orders: the ephemeral model is injected as a CTE named __dbt__cte__int_payments__successful

```bash
$ cat target/compiled/jaffle_dbt/models/marts/fct_orders.sql
```
```text
with

 __dbt__cte__int_payments__successful as (


with

payments as (

    select * from "practice"."dbt_adnane"."stg_stripe__payments"

),

successful as (

    select
        payment_id,
        order_id,
        payment_method,
        amount,
        created_at

    from payments
    where status = 'success'

)

select * from successful
), orders as (

    select * from "practice"."dbt_adnane"."stg_jaffle_shop__orders"

),

payments as (

    select * from __dbt__cte__int_payments__successful

),

order_payments as (

    select
        order_id,
        sum(amount) as amount

    from payments
    group by order_id

),

final as (

    select
        orders.order_id,
        orders.customer_id,
        orders.order_date,
        orders.status,
        coalesce(order_payments.amount, 0) as amount

    from orders
    left join order_payments
        on orders.order_id = order_payments.order_id

)

select * from final
```
