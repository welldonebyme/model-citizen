version: 2

snapshots:
  - name: customers_snapshot
    description: History of each customer's email and country (check strategy).
    relation: source('jaffle_shop', 'customers')
    config:
      schema: snapshots
      unique_key: id
      strategy: check
      check_cols: ['email', 'country']

  - name: orders_snapshot
    description: History of each order (timestamp strategy on _loaded_at).

    config:
      schema: snapshots
      unique_key: id
      strategy: timestamp
      updated_at: _loaded_at
