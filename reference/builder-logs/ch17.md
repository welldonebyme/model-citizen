## 2. Contract + constraints on dim_customers (models/marts/dim_customers.yml)

### models/marts/dim_customers.yml

```bash
$ cat models/marts/dim_customers.yml
```
```text
version: 2

models:
  - name: dim_customers
    description: One row per customer, with order history and lifetime value.
    config:
      tags: ['marketing']
      meta:
        contains_pii: true
      group: marketing
      access: public
      contract:
        enforced: true
    constraints:
      - type: check
        expression: "lifetime_value >= 0"
        name: lifetime_value_not_negative
    columns:
      - name: customer_id
        description: Primary key of the customer.
        data_type: integer
        constraints:
          - type: not_null
          - type: primary_key
      - name: first_name
        data_type: varchar
      - name: last_name
        data_type: varchar
      - name: email
        description: Email in lower case (NULL when unknown).
        data_type: varchar
        constraints:
          - type: unique
      - name: country
        data_type: varchar
      - name: region
        description: Sales region, from the country_regions seed.
        data_type: varchar
      - name: first_order_date
        data_type: date
      - name: most_recent_order_date
        data_type: date
      - name: number_of_orders
        data_type: bigint
        constraints:
          - type: not_null
      - name: lifetime_value
        description: Total of the successful payments, in dollars.
        data_type: decimal(38, 2)
        constraints:
          - type: not_null
```
