### .github/workflows/ci.yml

```bash

```
```text
# Slim CI for dbt Core + DuckDB.
# On every pull request to main:
#   1. build a fresh database with the raw data,
#   2. build PRODUCTION from the main branch to get its manifest (the state),
#   3. build only the models changed by the pull request (and their children)
#      in the schema dbt_ci, deferring every other ref to production.
name: dbt CI

on:
  pull_request:
    branches: [main]

jobs:
  slim-ci:
    runs-on: ubuntu-latest
    env:
      DBT_PROFILES_DIR: ${{ github.workspace }}/ci
      DBT_DUCKDB_PATH: ${{ github.workspace }}/ci_db/practice.duckdb
      DBT_USE_COLORS: "false"
    steps:
      - name: Check out the pull request (with the history, to find main)
        uses: actions/checkout@v4
        with:
          fetch-depth: 0

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: "3.12"

      - name: Install dbt
        run: pip install "dbt-core~=1.11.0" "dbt-duckdb~=1.11.0" "duckdb==1.5.6" pandas

      - name: Load the raw data into a fresh DuckDB file
        run: python ci/load_raw.py "$DBT_DUCKDB_PATH"

      - name: Build production from main (state to compare with)
        run: |
          git worktree add ../main origin/main
          cd ../main
          dbt deps
          dbt build --target prod --exclude-resource-type unit_test
          mkdir -p "$GITHUB_WORKSPACE/prod-artifacts"
          cp target/manifest.json "$GITHUB_WORKSPACE/prod-artifacts/"

      - name: Install packages
        run: dbt deps

      - name: List what the pull request changed
        run: dbt ls --select state:modified+ --state prod-artifacts --target ci

      - name: Create the changed relations, empty (unit tests need them to exist)
        run: dbt run --select state:modified+ --defer --state prod-artifacts --target ci --empty

      - name: Build and test the changes (unit tests, models, data tests)
        run: dbt build --select state:modified+ --defer --state prod-artifacts --target ci
```
